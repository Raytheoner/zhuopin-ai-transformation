"""OP-0913-B 一次性探针：重跑机制类可动 WIP 计入清单 + 状态格 claude/* 分支祖先核验。
只读；复用 工具-共享文档编辑锁.py::_mechanism_wip_counted_rows（同一套判据，不复制第二套）。"""
import importlib.util
import json
import re
import subprocess
from pathlib import Path

root = Path(__file__).resolve().parents[1]
spec = importlib.util.spec_from_file_location("lock", root / "0-学习与工具/工具-共享文档编辑锁.py")
m = importlib.util.module_from_spec(spec)
spec.loader.exec_module(m)
text = (root / m.QUEUE_MECHANISM_PATH_REL).read_text(encoding="utf-8")
secs = m._split_live_sections(text)
nums, degraded = m._mechanism_wip_counted_rows(secs.get("一", ""))
print("COUNT", len(nums))
print("ROWS", " ".join(nums))
for d in degraded:
    print("DEGRADED", d)
rows = {cells[0]: cells for _l, cells in m._table_data_rows(secs.get("一", ""))}


def git(*args):
    return subprocess.run(["git", *args], cwd=root, capture_output=True, text=True)


for n in nums:
    status = rows[n][5]
    br = sorted(set(re.findall(r"claude/[A-Za-z0-9._\-]+", status)))
    st = {}
    for b in br:
        if git("rev-parse", "--verify", "--quiet", b).returncode != 0:
            st[b] = "no-local-branch"
            continue
        st[b] = "in-master" if git("merge-base", "--is-ancestor", b, "master").returncode == 0 else "NOT-in-master"
    vals = list(st.values())
    tag = "none" if not vals else ("all" if all(v == "in-master" for v in vals) else ("partial" if any(v == "in-master" for v in vals) else "zero"))
    print(f"#{n}\t{tag}\t{json.dumps(st, ensure_ascii=False)}")
